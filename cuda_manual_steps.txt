# CUDA Installation Commands for WSL Ubuntu
# Run these commands one by one in your WSL Ubuntu terminal

# Step 1: Update packages
sudo apt update
sudo apt upgrade -y

# Step 2: Install prerequisites
sudo apt install -y build-essential git curl wget ca-certificates gnupg lsb-release

# Step 3: Add NVIDIA CUDA repository
wget https://developer.download.nvidia.com/compute/cuda/repos/wsl-ubuntu/x86_64/cuda-keyring_1.1-1_all.deb
sudo dpkg -i cuda-keyring_1.1-1_all.deb
rm cuda-keyring_1.1-1_all.deb
sudo apt update

# Step 4: Install CUDA Toolkit 12.4 (without drivers)
sudo apt install -y cuda-compiler-12-4 cuda-libraries-12-4 cuda-nvcc-12-4

# Step 5: Add CUDA to PATH (add these lines to ~/.bashrc)
echo 'export PATH=/usr/local/cuda-12.4/bin:$PATH' >> ~/.bashrc
echo 'export LD_LIBRARY_PATH=/usr/local/cuda-12.4/lib64:$LD_LIBRARY_PATH' >> ~/.bashrc
echo 'export CUDA_HOME=/usr/local/cuda-12.4' >> ~/.bashrc
source ~/.bashrc

# Step 6: Verify CUDA installation
nvcc --version
nvidia-smi

# Step 7: Install PyTorch with CUDA support
python3 -m pip install --upgrade pip
pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu124

# Step 8: Test PyTorch GPU
python3 -c "import torch; print('CUDA available:', torch.cuda.is_available()); print('GPU:', torch.cuda.get_device_name(0) if torch.cuda.is_available() else 'Not found')"
